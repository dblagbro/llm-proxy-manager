# v5.0.0 — Compliance Enforcement Design Spec

**Status:** locked through 33 architecture decisions; pending hub-team reply on sandbox/UA validation
**Target ship:** end of 2026-06-04 (24-48h from architecture close)
**Major version bump rationale:** scope is the largest single feature since v3.0.0 — new tables, new endpoints, new middleware, new sub-package (`app/compliance/`), changes to every dispatch path. Jumping v4.4 → v5.0 to signal the magnitude.

---

## 1. Overview

A per-API-key + per-deployment block-list system that prevents requests from reaching providers, models, or client products belonging to "blocked companies." Audit-grade enforcement with strict cluster-sync semantics, immutable audit trail, and substitution-with-disclosure as the primary user-facing behavior.

Driven by a US Government policy requirement on one of our consumer deployments. Today's block target is Anthropic; the architecture supports ANY company being banned by ANY deployment, with extensible custom-add for unknowns.

### Audit-defensible properties

- No request from a banned key ever reaches a banned provider, even as a fallback (HTTP 503 instead).
- No cached content or memory turn generated by a banned provider ever surfaces to a banned key.
- Every substitution, refusal, and policy change writes an immutable audit row.
- Banned client products (Claude CLI, Anthropic SDK, etc.) are refused with HTTP 451 regardless of which model the request asks for.
- Cluster-wide policy state is consistent within the cluster's sync window; policy changes require quorum (N-1 of N peers) before applying.

When the policy is first enabled on a deployment, run a one-shot migration:

```python
async def backfill_source_company_for_caller_memory(db):
    # Set source_company from join with providers.owner_company
    await db.execute(text("""
        UPDATE caller_memory
        SET source_company = (
            SELECT owner_company FROM providers
            WHERE providers.id = caller_memory.source_provider_id
        )
        WHERE source_company IS NULL
    """))
    # Anything still NULL stays NULL → treated as blocked by banned keys (safest)
    await db.commit()
```

Gated on `SystemSetting.compliance_backfill_applied=true` so it runs exactly once. Run for `caller_memory_marker` and `caller_memory` together.

Cache backfill: cache entries that lack `source_company` are simply not returned to banned keys (decision 7 unknown → blocked). They'll age out under normal cache TTL. No upfront migration needed.

```python
def provider_type_to_company(provider_type: str) -> Optional[str]:
    """Decision 17 — auto-derive owner_company at Provider create/update time."""
    for company_id, info in KNOWN_COMPANIES.items():
        if provider_type in info["provider_types"]:
            return company_id
    # Aggregator/relay types (openrouter, cursor-oauth) → return their OWN company
    if provider_type == "openrouter":
        return "openrouter"
    if provider_type == "cursor-oauth":
        return "cursor"
    return None

def model_family_to_company(model: str) -> Optional[str]:
    """Decision 11 — model-family is the source of truth for lineage."""
    if not model:
        return None
    m = model.lower()
    for company_id, info in KNOWN_COMPANIES.items():
        for prefix in info["model_prefixes"]:
            if m.startswith(prefix):
                return company_id
    return None
```

```python
class ComplianceNoSubstituteError(Exception):
    """Decision 4 — fires when filter eliminates all candidates."""

class ComplianceUaBlockedError(Exception):
    """Decision 16 — fires when banned UA detected."""
    def __init__(self, matched_product, matched_company, matched_pattern, audit_id):
        self.matched_product = matched_product
        self.matched_company = matched_company
        self.matched_pattern = matched_pattern
        self.audit_id = audit_id

_BLOCKLIST_CACHE: dict[str, tuple[float, set[str]]] = {}  # api_key_id → (expires_at, set)
_TTL_SEC = 30.0

async def get_effective_blocklist(db, api_key_id: str) -> set[str]:
    """Union of api_key.blocked_companies + SystemSetting.compliance_system_blocked_companies."""
    now = time.monotonic()
    cached = _BLOCKLIST_CACHE.get(api_key_id)
    if cached and cached[0] > now:
        return cached[1]
    # Fetch and union
    key = await get_api_key(db, api_key_id)
    per_key = set(key.blocked_companies or [])
    system = set(get_setting("compliance_system_blocked_companies", []) or [])
    effective = per_key | system
    _BLOCKLIST_CACHE[api_key_id] = (now + _TTL_SEC, effective)
    return effective

def invalidate_blocklist_cache(api_key_id: Optional[str] = None):
    """Called on PATCH /api/keys/{id} and PATCH /api/settings."""
    if api_key_id:
        _BLOCKLIST_CACHE.pop(api_key_id, None)
    else:
        _BLOCKLIST_CACHE.clear()

def filter_providers(providers, blocklist: set[str], requested_model: Optional[str] = None) -> list:
    """Decision 11+14+19 — drop providers whose owner_company is banned OR whose
    resolved model family is banned. Returns filtered list; raises
    ComplianceNoSubstituteError if empty + the original list wasn't empty."""
    if not blocklist:
        return providers
    out = []
    for p in providers:
        if p.owner_company in blocklist:
            continue
        # Family check via requested model OR provider's default model
        target_model = requested_model or p.default_model
        target_company = model_family_to_company(target_model)
        if target_company and target_company in blocklist:
            continue
        out.append(p)
    if not out and providers:
        raise ComplianceNoSubstituteError(
            f"All {len(providers)} candidates blocked by policy; banned companies: {blocklist}"
        )
    return out
```

### 4.4 `app/compliance/ua_detect.py`

```python
def detect_client_company(user_agent: str) -> Optional[tuple[str, str, str]]:
    """Returns (company_id, matched_pattern_value, matched_product_display) or None.
    Decision 16+22 — case-insensitive matching."""
    if not user_agent:
        return None
    ua = user_agent.lower()
    custom = get_custom_companies()
    for company_id, info in {**KNOWN_COMPANIES, **custom}.items():
        for rule in info["ua_patterns"]:
            pattern_type = rule["type"]
            value = rule["value"].lower()
            if pattern_type == "prefix" and ua.startswith(value):
                return (company_id, rule["value"], _product_label(value))
            elif pattern_type == "contains" and value in ua:
                return (company_id, rule["value"], _product_label(value))
            elif pattern_type == "regex" and re.search(rule["value"], user_agent, re.IGNORECASE):
                return (company_id, rule["value"], _product_label(rule["value"]))
            elif pattern_type == "exact" and ua == value:
                return (company_id, rule["value"], _product_label(value))
    return None
```

```python
def wants_sse_prelude(headers: dict) -> bool:
    """Decision 28 (revised 2026-06-03) — SSE prelude is OPT-IN.
    Default behavior is headers-only (protocol-clean SSE body).
    Caller sends Accept-Compliance-Events: true to opt in."""
    val = headers.get("accept-compliance-events", "").lower()
    return val == "true"

    """Decision 15 — emitted BEFORE message_start."""
    return (
        b"event: compliance_substitution\n"
        b"data: " + json.dumps(disclosure).encode() + b"\n\n"

    """Decision 15 — inject as top-level key on the first OpenAI frame."""
    first_data_frame["compliance_substitution"] = disclosure
    return first_data_frame
```

```python
async def emit_policy_change(
    db, *, scope: str, target_id: Optional[str], before: dict, after: dict,
    reason: str, changed_by_user_id: str,
    applied_to_peers: list, pending_peers: list,
):

    row = CompliancePolicyChange(

        before_state=json.dumps(before),
        after_state=json.dumps(after),
        reason=reason,
        applied_to_peers=json.dumps(applied_to_peers),
        pending_peers=json.dumps(pending_peers),
        cluster_sync_status=("fully-acked" if not pending_peers else f"quorum-reached-{len(pending_peers)}-pending"),
    )
    db.add(row)
    await db.flush()
    return policy_change_id

async def compute_daily_integrity_hash(db, day: datetime.date) -> str:
    """Decision 10 — soft immutability via daily hash chain."""
    # Get prior day's chain_hash
    prior = await db.execute(select(ComplianceAuditChain).where(
        ComplianceAuditChain.day == (day - timedelta(days=1)).isoformat()
    ))
    prior_row = prior.scalar_one_or_none()
    prior_hash = prior_row.chain_hash if prior_row else ""
    # Hash this day's events
    events = await db.execute(select(ComplianceEvent).where(
        func.date(ComplianceEvent.created_at) == day.isoformat()
    ).order_by(ComplianceEvent.id))
    rows = events.scalars().all()
    content = "".join(
        f"{r.id}|{r.audit_id}|{r.api_key_id}|{r.event_type}|{r.http_status}"
        for r in rows
    )
    chain_hash = hashlib.sha256((prior_hash + content).encode()).hexdigest()
    db.add(ComplianceAuditChain(
        day=day.isoformat(), row_count=len(rows),
        prior_day_chain_hash=prior_hash or None, chain_hash=chain_hash,
        computed_at=datetime.utcnow(),
    ))
    await db.commit()
    return chain_hash
```

```python
async def push_policy_change_with_quorum(
    self, payload: dict, required_acks: int, timeout_sec: float = 5.0,
) -> dict:
    """Decision 8 + 23 — fan out to all peers, return after quorum acks.
    Returns dict with applied_to_peers + pending_peers lists.
    Raises ClusterSyncQuorumNotReached if can't reach required_acks within timeout."""
    peers = self._active_peers()
    tasks = [self._push_to_peer(p, payload, timeout_sec) for p in peers]
    acks = []
    pending = []
    for task in asyncio.as_completed(tasks):
        try:
            peer, ack_time = await task
            acks.append({"peer": peer.name, "acked_at": ack_time.isoformat()})
            if len(acks) >= required_acks:
                # Quorum reached; collect remaining results without blocking
                for remaining in tasks:
                    if not remaining.done():
                        pending.append({"peer": remaining.peer.name, "reason": "lagging"})
                return {"applied_to_peers": acks, "pending_peers": pending,
                        "cluster_sync_status": "fully-acked" if not pending else f"quorum-reached-{len(pending)}-pending"}
        except Exception as e:
            pending.append({"peer": "?", "reason": str(e)})
    raise ClusterSyncQuorumNotReached(acks, pending)
```
